// ==UserScript==
// @name         圣诞帽粘贴助手 
// @name:en      Chaoxing Paste Assistant
// @match        *://*.chaoxing.com/*
// @grant        none
// @run-at       document-end
// ==/UserScript==

(function() {
    'use strict';

    // ==================== 1. 创建可拖拽UI ====================
    const div = document.createElement('div');
    div.id = 'chaoxing-paste-helper';
    div.style.cssText = `
        position:fixed; top:80px; right:20px; z-index:999999; 
        background:#fff; padding:12px; 
        border:1px solid #e0e0e0; border-radius:10px; 
        box-shadow:0 8px 24px rgba(0,0,0,0.12); 
        width:240px; font-family: -apple-system, BlinkMacSystemFont, "Segoe UI", Roboto, sans-serif;
        font-size:13px; color:#333;
        transition: box-shadow 0.2s;
        user-select: none;
    `;
    div.innerHTML = `
        <div id="drag-handle" style="
            padding:6px 10px; margin:-12px -12px 12px -12px; 
            background:linear-gradient(135deg, #667eea 0%, #764ba2 100%);
            border-radius:10px 10px 0 0; 
            cursor:move; font-size:13px; font-weight:600; 
            color:#fff; text-align:center; letter-spacing:0.5px;
        ">
            📋 粘贴助手
        </div>
        <div style="margin-bottom:8px;">
            <label style="display:block; font-size:12px; color:#888; margin-bottom:3px;">选择编辑器</label>
            <div style="display:flex; align-items:center; gap:6px;">
                <button id="prev-editor" style="
                    width:28px; height:28px; 
                    background:#f0f0f0; border:1px solid #ddd; border-radius:6px; 
                    cursor:pointer; font-size:16px; font-weight:bold; color:#555;
                    display:flex; align-items:center; justify-content:center;
                    flex-shrink:0;
                ">‹</button>
                <select id="editor-list" style="
                    flex:1; padding:6px 8px; 
                    border:1px solid #ddd; border-radius:6px; 
                    font-size:13px; background:#fafafa;
                    box-sizing:border-box;
                "></select>
                <button id="next-editor" style="
                    width:28px; height:28px; 
                    background:#f0f0f0; border:1px solid #ddd; border-radius:6px; 
                    cursor:pointer; font-size:16px; font-weight:bold; color:#555;
                    display:flex; align-items:center; justify-content:center;
                    flex-shrink:0;
                ">›</button>
            </div>
            <div id="editor-counter" style="font-size:11px; color:#aaa; text-align:center; margin-top:2px;"></div>
        </div>
        <div style="margin-bottom:8px;">
            <label style="display:block; font-size:12px; color:#888; margin-bottom:3px;">粘贴内容</label>
            <textarea id="paste-text" placeholder="在此输入或粘贴要注入的内容..." style="
                width:100%; height:80px; 
                border:1px solid #ddd; border-radius:6px; 
                padding:6px 8px; font-size:13px; 
                box-sizing:border-box; resize:vertical;
                font-family: inherit;
            "></textarea>
        </div>
        <button id="paste-btn" style="
            width:100%; padding:8px; 
            background:linear-gradient(135deg, #667eea 0%, #764ba2 100%);
            color:#fff; border:none; border-radius:6px; 
            font-size:14px; font-weight:600; cursor:pointer;
            transition: transform 0.1s, opacity 0.2s;
        ">✨ 注入到编辑器</button>
        <div id="editor-status" style="font-size:11px; color:#999; margin-top:4px; text-align:center;"></div>
    `;
    document.body.appendChild(div);

    // ==================== 2. 拖拽逻辑 ====================
    let isDragging = false;
    let offset = { x: 0, y: 0 };
    const handle = document.getElementById('drag-handle');

    handle.addEventListener('mousedown', (e) => {
        if (e.button === 0) {
            isDragging = true;
            offset.x = e.clientX - div.offsetLeft;
            offset.y = e.clientY - div.offsetTop;
            div.style.cursor = 'grabbing';
        }
    });

    document.addEventListener('mousemove', (e) => {
        if (isDragging) {
            let left = e.clientX - offset.x;
            let top = e.clientY - offset.y;
            left = Math.max(0, Math.min(left, window.innerWidth - div.offsetWidth));
            top = Math.max(0, Math.min(top, window.innerHeight - div.offsetHeight));
            div.style.left = left + 'px';
            div.style.top = top + 'px';
            div.style.right = 'auto';
        }
    });

    document.addEventListener('mouseup', () => {
        if (isDragging) {
            isDragging = false;
            div.style.cursor = 'default';
        }
    });

    // ==================== 3. 编辑器列表管理 ====================
    let editorCache = [];

    function getEditors() {
        const editors = [];

        // 检测 UEditor
        if (window.UE && window.UE.instants) {
            Object.keys(window.UE.instants).forEach(id => {
                const inst = window.UE.instants[id];
                if (inst && inst.body) {
                    editors.push({
                        type: 'ue',
                        id: id,
                        label: `UEditor: ${id}`,
                        instance: inst
                    });
                }
            });
        }

        // 检测 CodeMirror
        document.querySelectorAll('.CodeMirror').forEach((cmElem, index) => {
            const cm = cmElem.CodeMirror;
            if (cm) {
                const parent = cmElem.closest('.codemirror-wrap') || cmElem.parentElement;
                const label = parent?.querySelector('label')?.textContent?.trim() || 
                              cmElem.id || 
                              `CodeMirror-${index + 1}`;
                editors.push({
                    type: 'cm',
                    id: cmElem.id || `cm-${index}`,
                    label: `CodeMirror: ${label}`,
                    instance: cm,
                    element: cmElem
                });
            }
        });

        return editors;
    }

    function refreshList() {
        const select = document.getElementById('editor-list');
        const currentVal = select.value;
        const editors = getEditors();
        editorCache = editors;

        let optionsHtml = '<option value="">-- 请选择 --</option>';
        editors.forEach((ed, idx) => {
            const val = `${ed.type}:${ed.id}`;
            const isSelected = (val === currentVal) ? 'selected' : '';
            // 如果当前选中的值在列表里不存在，默认选第一个
            const isFirstAndNoMatch = (idx === 0 && !currentVal) ? 'selected' : '';
            const selected = isSelected || isFirstAndNoMatch;
            optionsHtml += `<option value="${val}" ${selected ? 'selected' : ''}>${ed.label}</option>`;
        });

        if (select.innerHTML !== optionsHtml) {
            select.innerHTML = optionsHtml;
        }

        // 更新计数器
        const counter = document.getElementById('editor-counter');
        const total = editors.length;
        const selectedIdx = editors.findIndex((ed, idx) => {
            const val = `${ed.type}:${ed.id}`;
            return val === select.value;
        });
        const displayIdx = selectedIdx >= 0 ? selectedIdx + 1 : 0;
        counter.textContent = total > 0 ? `${displayIdx} / ${total}` : '未发现编辑器';

        // 更新状态
        const statusDiv = document.getElementById('editor-status');
        statusDiv.textContent = total > 0 ? 
            `✅ 发现 ${total} 个编辑器` : 
            '⏳ 等待编辑器加载...';
        
        // 更新按钮状态
        document.getElementById('prev-editor').style.opacity = total > 0 ? '1' : '0.3';
        document.getElementById('next-editor').style.opacity = total > 0 ? '1' : '0.3';
        document.getElementById('prev-editor').style.cursor = total > 0 ? 'pointer' : 'default';
        document.getElementById('next-editor').style.cursor = total > 0 ? 'pointer' : 'default';
    }

    // ==================== 4. 左右切换逻辑 ====================
    function switchEditor(direction) {
        const select = document.getElementById('editor-list');
        const options = select.options;
        let currentIdx = select.selectedIndex;
        
        if (options.length <= 1) return; // 无有效选项
        
        let newIdx = currentIdx + direction;
        // 循环切换：到末尾跳回开头，到开头跳到末尾
        if (newIdx >= options.length) newIdx = 1; // 跳过第一个占位选项
        if (newIdx < 1) newIdx = options.length - 1;
        
        select.selectedIndex = newIdx;
        // 触发 change 事件，更新计数器
        select.dispatchEvent(new Event('change'));
    }

    document.getElementById('prev-editor').addEventListener('click', () => switchEditor(-1));
    document.getElementById('next-editor').addEventListener('click', () => switchEditor(1));

    // 监听select变化，更新计数器
    document.getElementById('editor-list').addEventListener('change', function() {
        const editors = getEditors();
        const total = editors.length;
        const selectedVal = this.value;
        const selectedIdx = editors.findIndex((ed) => {
            return `${ed.type}:${ed.id}` === selectedVal;
        });
        const displayIdx = selectedIdx >= 0 ? selectedIdx + 1 : 0;
        document.getElementById('editor-counter').textContent = total > 0 ? `${displayIdx} / ${total}` : '未发现编辑器';
    });

    // ==================== 5. 键盘快捷键切换 ====================
    document.addEventListener('keydown', (e) => {
        const select = document.getElementById('editor-list');
        // 当焦点在select上时，左右箭头自然切换，不需要额外处理
        if (document.activeElement === select) return;
        
        // Ctrl+Shift+[ : 上一个编辑器
        if (e.ctrlKey && e.shiftKey && e.key === '[') {
            e.preventDefault();
            switchEditor(-1);
        }
        // Ctrl+Shift+] : 下一个编辑器
        if (e.ctrlKey && e.shiftKey && e.key === ']') {
            e.preventDefault();
            switchEditor(1);
        }
    });

    // ==================== 6. 执行粘贴注入 ====================
    document.getElementById('paste-btn').addEventListener('click', function() {
        const val = document.getElementById('paste-text').value;
        const target = document.getElementById('editor-list').value;

        if (!target) {
            alert('⚠️ 请先选择一个编辑器！');
            return;
        }
        if (!val.trim()) {
            alert('⚠️ 请输入要粘贴的内容！');
            return;
        }

        const [type, id] = target.split(':');
        let success = false;

        if (type === 'ue') {
            const ue = window.UE.instants[id];
            if (ue) {
                try {
                    const html = val.replace(/\n/g, '<br>');
                    ue.setContent(html);
                    ue.fireEvent('contentChange');
                    success = true;
                } catch (e) {
                    console.error('UEditor注入失败:', e);
                }
            }
        } else if (type === 'cm') {
            const cmElem = document.getElementById(id)?.closest('.CodeMirror');
            const cm = cmElem?.CodeMirror;
            if (cm) {
                try {
                    const cursor = cm.getCursor();
                    cm.replaceRange(val, cursor);
                    success = true;
                } catch (e) {
                    console.error('CodeMirror注入失败:', e);
                }
            }
        }

        if (success) {
            const btn = document.getElementById('paste-btn');
            const origText = btn.textContent;
            btn.textContent = '✅ 注入成功！';
            btn.style.background = 'linear-gradient(135deg, #00b894 0%, #00cec9 100%)';
            setTimeout(() => {
                btn.textContent = origText;
                btn.style.background = '';
            }, 1500);
        } else {
            alert('❌ 注入失败，请确认编辑器已正确加载！');
        }
    });

    // ==================== 7. 快捷键唤起 ====================
    document.addEventListener('keydown', (e) => {
        if (e.ctrlKey && e.shiftKey && e.key === 'V') {
            e.preventDefault();
            div.style.display = 'block';
            document.getElementById('paste-text').focus();
            refreshList();
        }
        if (e.key === 'Escape' && div.style.display !== 'none') {
            div.style.display = 'none';
        }
    });

    // ==================== 8. 自动刷新 ====================
    let lastEditorCount = 0;
    setInterval(() => {
        const editors = getEditors();
        if (editors.length !== lastEditorCount) {
            lastEditorCount = editors.length;
            refreshList();
        }
        if (editors.length === 0) {
            document.getElementById('editor-status').textContent = '⏳ 正在等待编辑器加载...';
        }
    }, 2000);

    // ==================== 9. 初始加载 ====================
    setTimeout(refreshList, 500);

    console.log('📋 超星粘贴助手已启动！');
    console.log('   Ctrl+Shift+V  → 唤起助手');
    console.log('   Ctrl+Shift+[  → 上一个编辑器');
    console.log('   Ctrl+Shift+]  → 下一个编辑器');
    console.log('   ESC           → 关闭助手');
})();
